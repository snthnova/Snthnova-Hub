-- Function to create toggle buttons (with dynamic label sizing)
local function createToggleButton(name, position, stateKey)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, 0, 0, 45)
    frame.Position = UDim2.new(0, 0, position.Y.Scale, position.Y.Offset)
    frame.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    frame.BorderSizePixel = 0
    frame.ZIndex = 10
    frame.Parent = contentFrame
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = frame
    
    -- Label takes remaining space, toggle is 44px + 10px gap = 54px from right
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -64, 1, 0)  -- Full width minus space for toggle
    label.Position = UDim2.new(0, 10, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = name
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.TextSize = 14
    label.Font = Enum.Font.GothamMedium
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextTruncate = Enum.TextTruncate.AtEnd
    label.ZIndex = 10
    label.Parent = frame
    
    -- Toggle indicator
    local indicator = Instance.new("Frame")
    indicator.Size = UDim2.new(0, 44, 0, 22)
    indicator.Position = UDim2.new(1, -54, 0.5, -11)
    indicator.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
    indicator.BorderSizePixel = 0
    indicator.ZIndex = 10
    indicator.Parent = frame
    
    local indicatorCorner = Instance.new("UICorner")
    indicatorCorner.CornerRadius = UDim.new(0, 11)
    indicatorCorner.Parent = indicator
    
    local circle = Instance.new("Frame")
    circle.Size = UDim2.new(0, 18, 0, 18)
    circle.Position = UDim2.new(0, 2, 0.5, -9)
    circle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    circle.BorderSizePixel = 0
    circle.ZIndex = 11
    circle.Parent = indicator
    
    local circleCorner = Instance.new("UICorner")
    circleCorner.CornerRadius = UDim.new(0, 9)
    circleCorner.Parent = circle
    
    -- Button
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(1, 0, 1, 0)
    button.BackgroundTransparency = 1
    button.Text = ""
    button.ZIndex = 12
    button.Parent = frame
    
    -- Hover effect
    button.MouseEnter:Connect(function()
        frame.BackgroundColor3 = Color3.fromRGB(55, 55, 55)
    end)
    button.MouseLeave:Connect(function()
        frame.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    end)
    
    -- Toggle animation
    local function animateToggle()
        local targetPos = toggles[stateKey] and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
        local targetIndicatorColor = toggles[stateKey] and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(70, 70, 70)
        
        local tweenInfo = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        
        local circleTween = TweenService:Create(circle, tweenInfo, {Position = targetPos})
        local indicatorTween = TweenService:Create(indicator, tweenInfo, {BackgroundColor3 = targetIndicatorColor})
        
        circleTween:Play()
        indicatorTween:Play()
    end
    
    button.MouseButton1Click:Connect(function()
        toggles[stateKey] = not toggles[stateKey]
        animateToggle()
        print(name .. ": " .. (toggles[stateKey] and "ON" or "OFF"))
    end)
    
    return frame
end

-- Minimize/maximize functions
local function minimizeWindow()
    windowState.lastPosition = mainFrame.Position
    windowState.lastSize = mainFrame.Size
    windowState.isMinimized = true
    
    mainFrame.Visible = false
    minimizeBar.Visible = true
    minimizeBar.Position = windowState.lastPosition
end

local function maximizeWindow()
    windowState.isMinimized = false
    
    minimizeBar.Visible = false
    mainFrame.Visible = true
    mainFrame.Position = windowState.lastPosition or UDim2.new(0.5, -windowSize.X/2, 0.5, -windowSize.Y/2)
    mainFrame.Size = windowState.lastSize or UDim2.new(0, windowSize.X, 0, windowSize.Y)
end

minimizeButton.MouseButton1Click:Connect(function()
    if windowState.isMinimized then
        maximizeWindow()
    else
        minimizeWindow()
    end
end)

minimizeBarButton.MouseButton1Click:Connect(function()
    maximizeWindow()
end)

-- Dragging system
local UIS = UserInputService
local isDragging = false
local dragStart = nil
local startPos = nil
local dragObject = nil

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = true
        dragObject = mainFrame
        dragStart = input.Position
        startPos = mainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                isDragging = false
                dragObject = nil
            end
        end)
    end
end)

minimizeBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = true
        dragObject = minimizeBar
        dragStart = input.Position
        startPos = minimizeBar.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                isDragging = false
                dragObject = nil
            end
        end)
    end
end)

-- Resize system
local isResizing = false
local resizeStartPos = nil
local startSize = nil

resizeHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        isResizing = true
        resizeStartPos = input.Position
        startSize = mainFrame.AbsoluteSize
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                isResizing = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        local mousePos = input.Position
        
        if isDragging and dragObject then
            local delta = mousePos - dragStart
            dragObject.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
        
        if isResizing then
            local delta = mousePos - resizeStartPos
            local newWidth = math.clamp(startSize.X + delta.X, minSize.X, maxSize.X)
            local newHeight = math.clamp(startSize.Y + delta.Y, minSize.Y, maxSize.Y)
            
            mainFrame.Size = UDim2.new(0, newWidth, 0, newHeight)
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = false
        isResizing = false
        dragObject = nil
    end
end)
